BFSExample.java 
import java.util.*;

// BFS: Breadth-First Search 구현
public class BFSExample {
    public void bfs(List<List<Integer>> adj, int start) {
        // 1. 방문 여부를 저장할 배열 (중복 방문 방지)
        boolean[] visited = new boolean[adj.size()];

        // 2. 탐색을 위한 Queue (FIFO 구조)
        Queue<Integer> queue = new LinkedList<>();

        // 3. 시작 정점 예약
        visited[start] = true;
        queue.add(start);

        while (!queue.isEmpty()) {
            // 4. 큐에서 정점을 하나 꺼냄 (현재 위치)
            int curr = queue.poll();
            System.out.println("방문 정점: " + curr);

            // 5. 인접한 모든 정점을 확인
            for (int next : adj.get(curr)) {
                if (!visited[next]) {
                    // 6. 아직 방문하지 않은 경우만 큐에 추가
                    visited[next] = true;
                    queue.add(next);
                }
            }
        }
    }
}
